% ---------- Thirteenth Poem ----------
\poemtitle{A CHRISTMAS CAROL}
\addcontentsline{toc}{chapter}{A CHRISTMAS CAROL (Neapolitan)}

\begin{verse}
When Christ was born in Bethlehem,\\
'Twas night but seemed the noon of day:\\
\hspace*{2em}The star whose light\\
\hspace*{2em}Was pure and bright,\\
Shone with unwav'ring ray;\\
\hspace*{2em}But one bright star,\\
\hspace*{2em}One glorious star\\
Guided the Eastern Magi from afar.\\[0.7em]

Then peace was spread throughout the land;\\
The lion fed beside the lamb;\\
\hspace*{2em}And with the kid,\\
\hspace*{2em}To pastures led,\\
The spotted leopard fed\\
\hspace*{2em}In peace, in peace\\
\hspace*{2em}The calf and bear,\\
The wolf and lamb reposed together there.\\[0.7em]

As shepherds watched their flocks by night,\\
An angel brighter than the sun\\
\hspace*{2em}Appeared in air,\\
\hspace*{2em}And gently said,\\
"Fear not, be not afraid,\\
\hspace*{2em}Behold, behold,\\
\hspace*{2em}Beneath your eyes,\\
Earth has become a smiling Paradise."%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Translated from the Neapolitan}
