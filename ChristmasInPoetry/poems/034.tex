% ---------- Thirty-fourth Poem ----------
\poemhead{THE WASSAIL SONG}

\begin{verse}
Here we come a-wassailing\\
\hspace*{2em}Among the leaves so green,\\
Here we come a-wandering\\
\hspace*{2em}So fair to be seen.\\[0.7em]

Love and joy come to you\\
\hspace*{2em}And to your wassail too,\\
And God bless you, and send you\\
\hspace*{2em}A happy New Year.\\[0.7em]

We are not daily beggars\\
\hspace*{2em}That beg from door to door,\\
But we are neighbours' children\\
\hspace*{2em}That you have seen before.\\[0.7em]

Good Master and good Mistress,\\
\hspace*{2em}As you sit by the fire,\\
Pray think of us poor children\\
\hspace*{2em}Who are wandering in the mire.\\[0.7em]

Bring us out a table\\
\hspace*{2em}And spread it with a cloth;\\
Bring us out a mouldy cheese\\
\hspace*{2em}And some of your Christmas loaf.\\[0.7em]

God bless the master of this house,\\
\hspace*{2em}Likewise the mistress too;\\
And all the little children\\
\hspace*{2em}That round the table go.%
\end{verse}

\vspace{0.5\baselineskip}
