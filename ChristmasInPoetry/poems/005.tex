% ---------- Fifth Poem ----------
\poemtitle{BOOTS AND SADDLES}
\addcontentsline{toc}{chapter}{BOOTS AND SADDLES}

\begin{verse}
Our shepherds all\\
\hspace*{2em}As pilgrims have departed,\\
Our shepherds all\\
\hspace*{2em}Have gone to Bethlehem.\\
They gladly go\\
\hspace*{2em}For they are all stout-hearted,\\
They gladly go--\\
\hspace*{2em}Ah, could I go with them!\\[0.7em]

\hspace*{2em}I am too lame to walk,\\
\hspace*{4em}Boots and saddles, boots and saddles,\\
\hspace*{2em}I am too lame to walk,\\
\hspace*{4em}Boots and saddles, mount and ride.\\[0.7em]

A shepherd stout\\
\hspace*{2em}Who sang a catamiaulo,\\
A shepherd stout\\
\hspace*{2em}Was walking lazily.\\
He heard me speak\\
\hspace*{2em}And saw me hobbling after,\\
He turned and said\\
\hspace*{2em}He would give help to me.\\[0.7em]

"Here is my horse\\
\hspace*{2em}That flies along the high-road,\\
Here is my horse,\\
\hspace*{2em}The best in all the towns.\\
I bought him from\\
\hspace*{2em}A soldier in the army,\\
I got my horse\\
\hspace*{2em}By payment of five crowns."\\[0.7em]

When I have seen\\
\hspace*{2em}The Child, the King of Heaven,\\
When I have seen\\
\hspace*{2em}The Child who is God's son,\\
When to the mother,\\
\hspace*{2em}I my praise have given,\\
When I have finished,\\
\hspace*{2em}All I should have done:\\[0.7em]

\hspace*{2em}No more shall I be lame,\\
\hspace*{4em}Boots and saddles, boots and saddles,\\
\hspace*{2em}No more shall I be lame,\\
\hspace*{4em}Boots and saddles, mount and ride.%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Provençal Noël of Nicholas Saboly}\\
