% ---------- Thirty-second Poem ----------
\poemtitle{WE THREE KINGS}
\addcontentsline{toc}{chapter}{WE THREE KINGS}

\begin{verse}
We Three Kings of Orient are,\\
Bearing gifts we traverse afar,\\
\hspace*{2em}Field and fountain,\\
\hspace*{2em}Moor and mountain,\\
Following yonder star.\\[0.7em]

\textit{Chorus}\\
O Star of wonder, Star of night,\\
Star with Royal Beauty bright,\\
\hspace*{2em}Westward leading,\\
\hspace*{2em}Still proceeding,\\
Guide us to Thy perfect Light.\\[0.7em]

\textit{Gaspard:}\\
Born a king on Bethlehem plain,\\
Gold I bring to crown Him again;\\
\hspace*{2em}King forever,\\
\hspace*{2em}Ceasing never\\
Over us all to reign.\\[0.7em]

\textit{Chorus:} O Star of wonder\ldots\\[0.7em]

\textit{Melchior:}\\
Frankincense to offer have I,\\
Incense owns a deity nigh;\\
\hspace*{2em}Prayer and praising,\\
\hspace*{2em}All men raising,\\
Worship Him God on high.\\[0.7em]

\textit{Chorus:} O Star of wonder\ldots\\[0.7em]

\textit{Balthazar:}\\
Myrrh is mine; its bitter perfume\\
Breathes a life of gathering gloom;\\
\hspace*{2em}Sorrowing, sighing,\\
\hspace*{2em}Bleeding, dying,\\
Sealed in a stone-cold tomb.\\[0.7em]

\textit{Chorus:} O Star of wonder\ldots\\[0.7em]

Glorious now behold Him arise,\\
King and God, and Sacrifice;\\
\hspace*{2em}Heav'n sings Allelujah:\\
\hspace*{2em}Allelujah,\\
The earth replies.%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{J. H. Hopkins, Jr.}
