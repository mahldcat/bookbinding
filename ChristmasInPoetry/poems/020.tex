% ---------- Twentieth Poem ----------
\poemhead{NATIVITY SONG}

\begin{verse}
The beautiful mother is bending\\
\hspace*{2em}Low where her baby lies,\\
Helpless and frail, for her tending;\\
\hspace*{2em}But she knows the glorious eyes.\\[0.7em]

The mother smiles and rejoices\\
\hspace*{2em}While the baby laughs in the hay;\\
She listens to heavenly voices:\\
\hspace*{2em}"The child shall be king, one day."\\[0.7em]

O dear little Christ in the manger,\\
\hspace*{2em}Let me make merry with thee.\\
O King, in my hour of danger,\\
\hspace*{2em}Wilt thou be strong for me?%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Adapted from the Latin of Jacopone da Todi}\\
\noindent\textit{by Sophie Jewett}\\[0.5\baselineskip]
\noindent\textit{From \emph{The Poems of Sophie Jewett.} Included by permission of the Thomas Y. Crowell Company.}
