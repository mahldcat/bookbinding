% ---------- Eighteenth Poem ----------
\poemtitle{SONG OF A SHEPHERD BOY AT BETHLEHEM}
\addcontentsline{toc}{chapter}{SONG OF A SHEPHERD BOY AT BETHLEHEM}

\begin{verse}
Sleep, Thou little Child of Mary,\\
\hspace*{2em}Rest Thee now.\\
Though these hands be rough from shearing\\
\hspace*{2em}And the plow,\\
Yet they shall not ever fail Thee,\\
When the waiting nations hail Thee,\\
Bringing palms unto their King.\\
\hspace*{2em}Now--I sing.\\[0.7em]

Sleep, Thou little Child of Mary,\\
\hspace*{2em}Hope divine.\\
If Thou wilt but smile upon me,\\
\hspace*{2em}I will twine\\
Blossoms for Thy garlanding.\\
Thou'rt so little to be King,\\
\hspace*{2em}God's Desire!\\
\hspace*{2em}Not a brier\\
Shall be left to grieve Thy brow;\\
\hspace*{2em}Rest Thee now.\\[0.7em]

Sleep, Thou little Child of Mary,\\
\hspace*{2em}Some fair day\\
Wilt Thou, as Thou wert a brother,\\
\hspace*{2em}Come away\\
Over hills and over hollow?\\
All the lambs will up and follow.\\
Follow but for love of Thee.\\
\hspace*{2em}Lov'st Thou me?\\[0.7em]

Sleep, Thou little Child of Mary,\\
\hspace*{2em}Rest Thee now.\\
I that watch am come from sheep-stead\\
\hspace*{2em}And from plough.\\
Thou wilt have disdain of me\\
When Thou'rt lifted, royally,\\
Very high for all to see:\\
\hspace*{2em}Smilest Thou?%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Josephine Preston Peabody}\\
\noindent\textit{Included by permission of the author.}
