% ---------- Thirty-sixth Poem ----------
\poemhead{CAROL IN PRAISE OF THE HOLLY AND IVY}

\begin{verse}
Holly and Ivy made a great party,\\
\hspace*{2em}Who should have the mastery\\
\hspace*{4em}In lands where they go.\\[0.7em]

Then spake Holly, "I am fierce and jolly,\\
I will have the mastery\\
\hspace*{4em}In lands where we go."\\[0.7em]

Then spake Ivy, "I am loud and proud,\\
And I will have the mastery\\
\hspace*{4em}In lands where we go."\\[0.7em]

Then spake Holly, and bent him down on his knee,\\
"I pray thee, gentle Ivy,\\
Essay me no villany\\
\hspace*{4em}In the lands where we go."%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Fifteenth Century Carol}
