% ---------- Thirty-seventh Poem ----------
\poemtitle{CEREMONIES FOR CHRISTMAS}
\addcontentsline{toc}{chapter}{CEREMONIES FOR CHRISTMAS}

\begin{verse}
\hspace*{2em}Come, bring with a noise,\\
\hspace*{2em}My merry, merry boys,\\
The Christmas log to the firing,\\
\hspace*{2em}While my good dame, she\\
\hspace*{2em}Bids ye all be free,\\
And drink to your heart's desiring.\\[0.7em]

\hspace*{2em}With the last year's brand\\
\hspace*{2em}Light the new block, and\\
For good success in his spending,\\
\hspace*{2em}On your psalteries play,\\
\hspace*{2em}That sweet luck may\\
Come while the log is a-tending.\\[0.7em]

\hspace*{2em}Drink now the strong beer,\\
\hspace*{2em}Cut the white loaf here,\\
The while the meat is a-shredding;\\
\hspace*{2em}For the rare mince-pie\\
\hspace*{2em}And the plums stand by\\
To fill the paste that's a-kneading.%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Robert Herrick}
