% ---------- Twenty-first Poem ----------
\poemtitle{THE CHRISTMAS SILENCE}
\addcontentsline{toc}{chapter}{THE CHRISTMAS SILENCE}

\begin{verse}
Hushed are the pigeons cooing low,\\
\hspace*{2em}On dusty rafters of the loft;\\
\hspace*{2em}And mild-eyed oxen, breathing soft,\\
Sleep on the fragrant hay below.\\[0.7em]

Dim shadows in the corner hide;\\
\hspace*{2em}The glimmering lantern's rays are shed\\
\hspace*{2em}Where one young lamb just lifts his head,\\
Then huddles 'gainst his mother's side.\\[0.7em]

Strange silence tingles in the air;\\
\hspace*{2em}Through the half-open door a bar\\
\hspace*{2em}Of light from one low hanging star\\
Touches a baby's radiant hair--\\[0.7em]

No sound--the mother, kneeling, lays\\
\hspace*{2em}Her cheek against the little face.\\
\hspace*{2em}Oh human love! Oh heavenly grace!\\
'Tis yet in silence that she prays!\\[0.7em]

Ages of silence end to-night;\\
\hspace*{2em}Then to the long-expectant earth\\
\hspace*{2em}Glad angels come to greet His birth\\
In burst of music, love, and light!%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Margaret Deland}\\
