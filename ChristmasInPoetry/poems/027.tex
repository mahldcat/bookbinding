% ---------- Twenty-seventh Poem ----------
\poemtitle{CAROL}
\addcontentsline{toc}{chapter}{CAROL (Canton)}

\begin{verse}
When the herds were watching\\
\hspace*{2em}In the midnight chill,\\
Came a spotless lambkin\\
\hspace*{2em}From the heavenly hill.\\[0.7em]

Snow was on the mountains,\\
\hspace*{2em}And the wind was cold,\\
When from God's own garden\\
\hspace*{2em}Dropped a rose of gold.\\[0.7em]

When 'twas bitter winter,\\
\hspace*{2em}Houseless and forlorn\\
In a star-lit stable\\
\hspace*{2em}Christ the Babe was born.\\[0.7em]

Welcome, heavenly lambkin;\\
\hspace*{2em}Welcome, golden rose;\\
Alleluia, Baby,\\
\hspace*{2em}In the swaddling clothes!%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{William Canton}
