% ---------- Third Poem ----------
\poemhead{LORDINGS, LISTEN TO OUR LAY}

\begin{verse}
Lordings, listen to our lay--\\
We have come from far away\\
\hspace*{2em}To seek Christmas;\\
In this mansion we are told\\
He His yearly feast doth hold:\\
\hspace*{2em}'Tis to-day!\\
May joy come from God above,\\
To all those who Christmas love.%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Old Carol}
