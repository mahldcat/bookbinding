% ---------- Thirty-eighth Poem ----------
\poemtitle{CHRISTMAS EVE--ANOTHER CEREMONY}
\addcontentsline{toc}{chapter}{CHRISTMAS EVE--ANOTHER CEREMONY}

\begin{verse}
Come, guard this night the Christmas-pie,\\
That the thief, though ne'er so sly,\\
With his flesh-hooks, don't come nigh\\
\hspace*{10em}To catch it.\\[0.7em]

From him, who alone sits there,\\
Having his eyes still in his ear,\\
And a deal of nightly fear\\
\hspace*{10em}To watch it.%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Robert Herrick}
