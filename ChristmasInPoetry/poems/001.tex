\poemtitle{\normalsize A CHRISTMAS CAROL}
\addcontentsline{toc}{chapter}{A CHRISTMAS CAROL}

\begin{verse}
God bless the master of this house,\\
\hspace*{2em}The mistress also,\\
And all the little children,\\
\hspace*{2em}That round the table go.\\[0.7em]

And all your kin and folk,\\
\hspace*{2em}That dwell both far and near;\\
I wish you a merry Christmas,\\
\hspace*{2em}And a happy New Year.%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Attributed to: Old English Carol}
