% ---------- Twenty-sixth Poem ----------
\poemtitle{A CHRISTMAS CAROL}
\addcontentsline{toc}{chapter}{A CHRISTMAS CAROL (Rossetti)}

\begin{verse}
In the bleak mid-winter\\
\hspace*{2em}Frosty wind made moan,\\
Earth stood hard as iron,\\
\hspace*{2em}Water like a stone;\\
Snow had fallen, snow on snow,\\
\hspace*{2em}Snow on snow,\\
In the bleak mid-winter\\
\hspace*{2em}Long ago.\\[0.7em]

Our God, Heaven cannot hold Him\\
\hspace*{2em}Nor earth sustain;\\
Heaven and earth shall flee away\\
\hspace*{2em}When He comes to reign.\\
In the bleak mid-winter\\
\hspace*{2em}A stable-place sufficed\\
The Lord God Almighty\\
\hspace*{2em}Jesus Christ.\\[0.7em]

Angels and archangels\\
\hspace*{2em}May have gathered there,\\
Cherubim and seraphim\\
\hspace*{2em}Thronged the air;\\
But only His Mother\\
\hspace*{2em}In her maiden bliss\\
Worshipped her Beloved\\
\hspace*{2em}With a kiss.\\[0.7em]

What can I give Him,\\
\hspace*{2em}Poor as I am?\\
If I were a shepherd\\
\hspace*{2em}I would bring a lamb,\\
If I were a Wise Man,\\
\hspace*{2em}I would do my part,--\\
Yet what I can I give Him,\\
\hspace*{2em}Give my heart.%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Christina G. Rossetti}
