% ---------- Twenty-fourth Poem ----------
\poemtitle{AS JOSEPH WAS A-WALKING}
\addcontentsline{toc}{chapter}{AS JOSEPH WAS A-WALKING}

\begin{verse}
As Joseph was a-walking\\
\hspace*{2em}He heard an angel sing:--\\
"This night there shall be born\\
\hspace*{2em}Our heavenly King.\\[0.7em]

"He neither shall be born\\
\hspace*{2em}In housen, nor in hall,\\
Nor in the place of Paradise,\\
\hspace*{2em}But in an ox's stall.\\[0.7em]

"He neither shall be clothéd\\
\hspace*{2em}In purple nor in pall;\\
But in the fair, white linen,\\
\hspace*{2em}That usen babies all.\\[0.7em]

"He neither shall be rockéd\\
\hspace*{2em}In silver nor in gold,\\
But in a wooden cradle\\
\hspace*{2em}That rocks on the mould.\\[0.7em]

"He neither shall be christened\\
\hspace*{2em}In white wine nor in red,\\
But with fair spring water\\
\hspace*{2em}With which we were christenéd."\\[0.7em]

Mary took her baby,\\
\hspace*{2em}She dressed Him so sweet,\\
She laid Him in a manger,\\
\hspace*{2em}All there for to sleep.\\[0.7em]

As she stood over Him\\
\hspace*{2em}She heard angels sing,\\
"O bless our dear Saviour,\\
\hspace*{2em}Our heavenly King."%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{From the Cherry Tree Carol}
