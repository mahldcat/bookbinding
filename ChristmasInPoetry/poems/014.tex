% ---------- Fourteenth Poem ----------
\poemtitle{THE GOLDEN CAROL}
\addcontentsline{toc}{chapter}{THE GOLDEN CAROL}

\begin{center}
\textit{(Of Melchior, Balthazar, and Caspar, the Three Kings)}
\end{center}

\begin{verse}
We saw the light shine out a-far,\\
\hspace*{2em}On Christmas in the morning.\\
And straight we knew Christ's Star it was,\\
Bright beaming in the morning.\\
Then did we fall on bended knee,\\
On Christmas in the morning,\\
And prais'd the Lord, who'd let us see\\
His glory at its dawning.\\[0.7em]

Oh! every thought be of His Name,\\
\hspace*{2em}On Christmas in the morning,\\
Who bore for us both grief and shame,\\
\hspace*{2em}Afflictions sharpest scorning.\\
And may we die (when death shall come),\\
\hspace*{2em}On Christmas in the morning,\\
And see in heav'n, our glorious home,\\
\hspace*{2em}The Star of Christmas morning.%
\end{verse}
