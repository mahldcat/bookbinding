% ---------- Thirty-ninth Poem ----------
\poemtitle{ANOTHER TO THE MAIDS}
\addcontentsline{toc}{chapter}{ANOTHER TO THE MAIDS}

\begin{verse}
Wash your hands, or else the fire\\
Will not tend to your desire;\\
Unwashed hands, ye maidens, know,\\
Dead the fire, though ye blow.%
\end{verse}

\vspace{0.5\baselineskip}
\noindent\textit{Robert Herrick}
